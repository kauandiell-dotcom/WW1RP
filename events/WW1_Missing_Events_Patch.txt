# Generic decision event 1: recruit general for civil war
country_event = {
	id = decisions.1
	title = decisions.1.t
	desc = decisions.1.desc
	picture = GFX_report_event_generic_handshake

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			random_army_leader = {
				limit = {
					is_field_marshal = no
					NOT = { has_unit_leader_flag = hidden_sympathies }
				}
				set_unit_leader_flag = hidden_sympathies
			}
		}
	}

	option = {
		name = decisions.1.a
	}
}
